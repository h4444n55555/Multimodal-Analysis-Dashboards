\newcommand{\cbox}{$\square$}

\newcommand{\writeline}{\par\vspace{7pt}\fillline{\linewidth}}

% one task block
\newcommand{\task}[3]{%
  \textbf{#1} \hfill \small #2\normalsize\par\vspace{2pt}
  {\small\itshape #3}\par\vspace{5pt}
  Outcome:\quad \cbox~Completed \quad \cbox~Completed with help \quad \cbox~Failed
  \hfill Time: \fillline{1.4cm}\,s \quad Errors: \fillline{1cm}\par\vspace{6pt}
  Where they got confused / what they said:
  \writeline\writeline\writeline\writeline\par\vspace{10pt}}

% one SUS row: statement + 1–5 boxes
\newcommand{\sus}[2]{#1 & #2 & \cbox & \cbox & \cbox & \cbox & \cbox \\}

\newcommand{\sheet}[1]{%
  {\Large\textbf{User test --- #1}}\hfill Date: \fillline{2.6cm}\par\vspace{8pt}
  Background: \cbox~Non-technical \quad \cbox~Technical / research student \quad
  \cbox~Other: \fillline{3cm}\par\vspace{3pt}
  Used a similar health or data app before? \quad \cbox~Yes \quad \cbox~No\par
  \vspace{10pt}\hrule\vspace{10pt}

  \task{T1 --- Aura Screen (phone)}{Success: reads their heart rate on the report}
  {Create an account, finish setting up your record, start a contactless scan, then open
   your session report and tell us your heart rate.}

  \task{T2 --- Study Hub website (laptop)}{Success: states both heart rates + reaches the agreement step}
  {Find the ECG data. How does heart rate differ between resting and city driving?
   Then start a request to access the dataset.}

  \hrule\vspace{8pt}
  \textbf{After the tasks --- SUS} \hfill {\small 1 = strongly disagree \quad 5 = strongly agree}\par
  \vspace{2pt}
  {\small
  \begin{tabular}{@{}r p{12.2cm} ccccc@{}}
  \toprule
   & & 1 & 2 & 3 & 4 & 5 \\
  \midrule
  \sus{1}{I think that I would like to use this system frequently.}
  \sus{2}{I found the system unnecessarily complex.}
  \sus{3}{I thought the system was easy to use.}
  \sus{4}{I think that I would need the support of a technical person to use this system.}
  \sus{5}{I found the various functions in this system were well integrated.}
  \sus{6}{I thought there was too much inconsistency in this system.}
  \sus{7}{I would imagine that most people would learn to use this system very quickly.}
  \sus{8}{I found the system very cumbersome to use.}
  \sus{9}{I felt very confident using the system.}
  \sus{10}{I needed to learn a lot of things before I could get going with this system.}
  \bottomrule
  \end{tabular}}
  \par\vspace{8pt}
  \textbf{Debrief}\par\vspace{2pt}
